GET/v1/images/generations/{id}
Python
import time, requests

BASE = "https://muse.isemi.io/v1"
HEADERS = {"Authorization": "Bearer m2a-your-key"}

task = requests.post(f"{BASE}/images/generations", headers=HEADERS, json={
    "prompt": "A red fox sitting in an autumn forest",
    "async": True,
}).json()

while task["status"] in ("queued", "running"):
    time.sleep(5)
    task = requests.get(f"{BASE}/images/generations/{task['id']}", headers=HEADERS).json()

if task["status"] == "succeeded":
    print(task["result"]["data"][0]["url"])
else:
    print("Error:", task["error"])

import time, requests

BASE = "https://muse.isemi.io/v1"
HEADERS = {"Authorization": "Bearer m2a-your-key"}

task = requests.post(f"{BASE}/videos", headers=HEADERS, json={
    "prompt": "Maple leaves falling in a light breeze",
    "duration": 10,
    "size": "16:9",
}).json()

while task["status"] in ("queued", "running"):
    time.sleep(10)
    task = requests.get(f"{BASE}/videos/{task['id']}", headers=HEADERS).json()
    print(task["status"], task["progress"])

print(task["result"]["url"] if task["status"] == "succeeded" else task["error"])
